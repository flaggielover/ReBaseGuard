def git(d, *a, inp=None, check=True):
    r = subprocess.run(["git", "-C", str(d), *CFG, *a], capture_output=True, input=inp)
    if check and r.returncode:
        raise RuntimeError(f"git {a}: {r.stderr.decode()[-400:]}")
    return r.stdout.decode().strip()


def init(d):
    Path(d).mkdir(parents=True, exist_ok=True)
    git(d, "init", "-q")
    for k, v in (("gc.auto", "0"), ("core.hooksPath", os.devnull), ("maintenance.auto", "false")):
        git(d, "config", "--local", k, v)


def commit(d, files=None, rm=()):
    for rel, data in (files or {}).items():
        p = Path(d, rel)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(data)
    for rel in rm:
        git(d, "rm", "-q", rel)
    git(d, "add", "-A")
    git(d, "commit", "-q", "--allow-empty", "-m", "c")
    return git(d, "rev-parse", "HEAD")


def valid(d, path="att/HOST.json", sha=SHA):
    try:
        rec, why = L.attestation_valid({"path": path, "sha256": sha}, repo_root=str(d))
        return {"valid": rec is not None, "reasons": why}
    except Exception as e:
        return {"valid": None, "exception": f"{type(e).__name__}: {e}"}


def hist(d):
    try:
        return L._history_complete(str(d))
    except Exception as e:
        return f"EXC {type(e).__name__}: {e}"


def raw(d, *a):
    r = subprocess.run(["git", "-C", str(d), *L._GIT_ISOLATION, *a], capture_output=True, env=L._git_env())
    return {"rc": r.returncode, "out": r.stdout.decode(errors="replace")[-600:], "err": r.stderr.decode(errors="replace")[-300:]}


def mkcommit(d, tree, parents, msg="c", extra_headers=""):
    body = f"tree {tree}\n" + "".join(f"parent {p}\n" for p in parents)
    body += "author t <t@t> 1700000000 +0000\ncommitter t <t@t> 1700000000 +0000\n" + extra_headers + f"\n{msg}\n"
    return git(d, "hash-object", "-t", "commit", "-w", "--stdin", inp=body.encode())


def mktree_literal(d, entries):
    """entries: list of (mode, name, oid) already sorted by the caller; written with --literally (no checks)."""
    data = b"".join(m.encode() + b" " + n.encode() + b"\0" + bytes.fromhex(o) for m, n, o in entries)
    return git(d, "hash-object", "-t", "tree", "-w", "--literally", "--stdin", inp=data)


def probe(name):
    def deco(fn):
        try:
            with tempfile.TemporaryDirectory(dir=TMP) as w:
                results[name] = fn(Path(w))
        except Exception:
            results[name] = {"probe_error": traceback.format_exc()[-1500:]}
        return fn
    return deco

GPG_SIG = ("gpgsig -----BEGIN PGP SIGNATURE-----\n \n iQEzBAABCAAdFiEEabcdefabcdefabcdefabcdefabcdefabcd\n"
           " -----END PGP SIGNATURE-----\n")

def signed_child(d, parent, files):
    """A commit with `files` on top of `parent` (via the index), carrying a (fake) gpgsig header."""
    for rel, data in files.items():
        p = Path(d, rel)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(data)
    git(d, "add", "-A")
    tree = git(d, "write-tree")
    c = mkcommit(d, tree, [parent], extra_headers=GPG_SIG)
    git(d, "update-ref", "HEAD", c)
    return c




def wt_read(d, commit_id, rel):
    """What a checkout of commit_id shows at rel (follows symlinks like a reader would)."""
    wt = Path(str(d) + "_wt_" + commit_id[:8])
    git(d, "worktree", "add", "-q", "--detach", str(wt), commit_id)
    try:
        p = wt / rel
        return {"exists": p.exists(), "is_symlink": p.is_symlink(),
                "text": p.read_text()[:200] if p.exists() and p.is_file() else None}
    finally:
        git(d, "worktree", "remove", "--force", str(wt))


def fsck(d):
    a = raw(d, "fsck", "--full", "--strict", "--no-dangling", "--no-progress")
    return {"rc": a["rc"], "err_tail": a["err"][-200:]}

def n1_source(d):
    init(d)
    c1 = commit(d, {"att/HOST.json": ATT})
    blob = git(d, "hash-object", "-w", "--stdin", inp=VETO)
    att_tree = git(d, "rev-parse", "HEAD:att")
    entries = sorted([("40000", "att", att_tree), ("100644", f"{RD}/veto.md", blob)], key=lambda e: e[1] + ("/" if e[0] == "40000" else ""))
    tree = mktree_literal(d, entries)
    c2 = mkcommit(d, tree, [c1], msg="veto")
    c3 = mkcommit(d, git(d, "rev-parse", f"{c1}^{{tree}}"), [c2], msg="rm")
    git(d, "update-ref", "HEAD", c3); git(d, "reset", "-q", "--hard")
    commit(d, {f"{ACC}1.md": OK})
    return c2


@probe("n1_in_alternate_shared_clone")

@probe("n1_with_fsck_key_downgrade")
def _(w):
    out = {}
    for scope in ("local", "worktree", "include"):
        d = w / scope
        n1_source(d)
        if scope == "local":
            git(d, "config", "--local", "fsck.fullPathname", "ignore")
        elif scope == "worktree":
            git(d, "config", "--local", "extensions.worktreeConfig", "true")
            git(d, "config", "--worktree", "fsck.fullPathname", "ignore")
        else:
            Path(d, ".git/x.cfg").write_text("[FSCK]\n\tFullPathName = ignore\n")
            git(d, "config", "--local", "include.path", "x.cfg")
        out[scope] = {"plain_fsck_strict_rc": subprocess.run(["git", "-C", str(d), "fsck", "--strict"], capture_output=True).returncode,
                      "history_complete": hist(d), "result": valid(d)}
    return out

@probe("configured_command_sweep")
def _(w):
    marker = w / "m"
    script = w / "cmd.sh"
    script.write_text(f"#!/bin/sh\necho \"$0 $*\" >> '{marker}'\ncat >/dev/null 2>&1\nexit 0\n"); script.chmod(0o755)
    hooks = w / "hooks"; hooks.mkdir()

              "commit-msg", "post-commit", "pre-rebase", "post-checkout", "post-merge", "pre-push", "pre-receive", "update",

              "p4-post-changelist", "p4-pre-submit", "post-index-change"):
        os.symlink(script, hooks / h)
    out = {}
    for case in ("valid", "rejected_order_rule"):
        d = w / case; init(d)
        if case == "valid":
            c0 = commit(d, {"README": b"x\n", ".gitattributes": b"* diff=x filter=x merge=x\n"})
            signed_child(d, c0, {"att/HOST.json": ATT})

        else:                                     # quote before blob: the order rule must refuse
            commit(d, {"notes/quote.md": f"{LINE}\n".encode(), ".gitattributes": b"* diff=x filter=x merge=x\n"})
            c1 = git(d, "rev-parse", "HEAD")
            signed_child(d, c1, {"att/HOST.json": ATT})
            commit(d, {f"{ACC}1.md": OK})
        before = valid(d)
        s = str(script)
        for k, v in (("core.pager", s), ("pager.log", s), ("pager.config", s), ("pager.fsck", s), ("core.editor", s),
                     ("sequence.editor", s), ("core.sshCommand", s), ("core.askPass", s), ("credential.helper", s),
                     ("gpg.program", s), ("gpg.openpgp.program", s), ("gpg.ssh.program", s), ("gpg.x509.program", s),
                     ("gpg.ssh.defaultKeyCommand", s), ("log.showSignature", "true"), ("diff.external", s),
                     ("diff.x.textconv", s), ("diff.x.command", s), ("filter.x.clean", s), ("filter.x.smudge", s),
                     ("filter.x.process", s), ("merge.x.driver", s), ("core.hooksPath", str(hooks)),

        if marker.exists():
