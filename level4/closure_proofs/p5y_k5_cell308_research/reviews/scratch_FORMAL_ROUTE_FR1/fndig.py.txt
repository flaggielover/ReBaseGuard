import ast, hashlib, sys, json
path, names = sys.argv[1], sys.argv[2].split(',')
src = open(path, encoding='utf-8').read()
tree = ast.parse(src)
lines = src.splitlines(keepends=True)
out = {}
for node in tree.body:
    nm = None
    if isinstance(node, (ast.FunctionDef, ast.ClassDef, ast.AsyncFunctionDef)):
        nm = node.name
    elif isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
        nm = node.targets[0].id
    elif isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
        nm = node.target.id
    if nm in names:
        seg = ast.get_source_segment(src, node)
        start = node.lineno
        if hasattr(node, 'decorator_list') and node.decorator_list:
            start = min(d.lineno for d in node.decorator_list)
        out[nm] = dict(lines=f"{start}-{node.end_lineno}", seg_sha=hashlib.sha256(seg.encode()).hexdigest(),
                       lines_sha=hashlib.sha256(''.join(lines[start-1:node.end_lineno]).encode()).hexdigest())
print(json.dumps(out, indent=1))
