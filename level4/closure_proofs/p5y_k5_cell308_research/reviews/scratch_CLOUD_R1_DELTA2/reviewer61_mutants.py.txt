def run_suite(d, tests_dir=TESTS):
    env = dict(os.environ, LX_PLATFORM_DIR=d, PYTHONDONTWRITEBYTECODE="1")
    p = subprocess.run([sys.executable, "-B", "-m", "unittest", "discover", "-s", str(tests_dir)], cwd=str(tests_dir.parent),
                       env=env, capture_output=True, text=True, timeout=900)
    fails = sorted({ln.split()[1] for ln in p.stderr.splitlines() if ln.startswith(("FAIL:", "ERROR:"))})
    return p.returncode, (p.stderr.strip().splitlines() or [""])[-1], fails

rows = []

for mid, what, edits in MUTANTS:
    m = src
    for old, new in edits:
        assert m.count(old) == 1, (mid, old)
        m = m.replace(old, new)
    diff = "".join(difflib.unified_diff(src.splitlines(True), m.splitlines(True), n=0))
    diff = "".join(l for l in diff.splitlines(True) if not l.startswith(("---", "+++")))
    with tempfile.TemporaryDirectory(dir=scratch) as d:
        Path(d, "lx_platform.py").write_text(m)
        rc, tail, fails = run_suite(d)
    rows.append({"id": mid, "property": what, "status": "KILLED" if rc else "SURVIVED", "tail": tail, "killed_by": fails,
                 "diff": diff})
