from pathlib import Path
HERE = Path(__file__).resolve().parent
spec = importlib.util.spec_from_file_location("m", HERE / "reviewer61_mutants.py")
src = (HERE / "base" / "lx_platform.py").read_text()

exec("MUTANTS = " + txt.split("MUTANTS = ", 1)[1].split("\ndef run_suite")[0], ns)
def load(code, name):
    d = tempfile.mkdtemp(dir=HERE / "mut_tmp"); Path(d, "lx_platform.py").write_text(code)
    s = importlib.util.spec_from_file_location(name, Path(d, "lx_platform.py")); m = importlib.util.module_from_spec(s)
    s.loader.exec_module(m); return m
def mutant(mid):
    m = src
    for o, n in next(e for i, _, e in ns["MUTANTS"] if i == mid):
        m = m.replace(o, n)
    return load(m, mid)
B = load(src, "base")
HW = {"machine_id_sha256": "a" * 64, "product_uuid_sha256": "b" * 64, "hardware_bound": True}
def repo(d, files):
    subprocess.run(["git", "init", "-q", d], check=True)
    for rel, data in files.items():
        Path(d, rel).parent.mkdir(parents=True, exist_ok=True); Path(d, rel).write_bytes(data)
    subprocess.run(["git", "-C", d, "add", "-A"], check=True)
    subprocess.run(["git", "-C", d, "-c", "user.name=t", "-c", "user.email=t@t", "-c", "commit.gpgsign=false", "commit", "-q", "-m", "a"], check=True)
def att(L, **o):
    r = {"schema": L.ATTESTATION_SCHEMA, "reviewed": True, "accepted": True, "machine": HW, "acceptance_record": "reviews/A.md",
         "provider_image_updates": L.NO_IMAGE_CHANGE}; r.update(o); return json.dumps(r).encode()
out = {}

    return L.preflight("/", mode="launcher_preflight", other_job_running=False, launched=None, attestation_ref=None, decided=decided,

                       attestation_check=lambda r: ({"provider_image_updates": L.NO_IMAGE_CHANGE}, []))["gates"]["host_lifetime_attested"]
