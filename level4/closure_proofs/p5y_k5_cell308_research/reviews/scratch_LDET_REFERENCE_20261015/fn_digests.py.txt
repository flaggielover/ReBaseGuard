"""Parse-only digests of named top-level definitions in candidate files (git show; nothing imported or executed)."""
import ast, hashlib, json, subprocess, sys
REPO = "/home/user/ReBaseGuard"
C = "142d97a90a1c3fafff394044aa1448674f7a633c"
BASE = "level4/closure_proofs/p5y_k5_cell308_mbs_r1/code/"
WANT = {"mbs308_qualify.py": ["TIMING_KEYS", "MBR1_RECORD_PINS", "pinned_bytes", "strip_timing", "_strip_decoy_timing",
                              "_leaves", "_public", "_serial_view", "_ladder_view", "_decoy_view", "_compare",
                              "_planted", "_job_rows", "mbr1_repro"],
        "mbs308_driver.py": ["fs", "jsonable"]}
out = {"commit": C, "method": "sha256 of ast.get_source_segment (exact source text) and of ast.dump(node) "
       "(include_attributes=False); parse only", "files": {}}
for f, names in WANT.items():
    rel = BASE + f
    src = subprocess.run(["git", "-C", REPO, "show", f"{C}:{rel}"], capture_output=True, check=True).stdout
    blob = subprocess.run(["git", "-C", REPO, "rev-parse", f"{C}:{rel}"], capture_output=True, text=True,
                          check=True).stdout.strip()
    text = src.decode()
    tree = ast.parse(text)
    defs = {}
    for node in tree.body:
        nm = None
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            nm = node.name
        elif isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            nm = node.targets[0].id
        if nm in names:
            assert nm not in defs, (f, nm)
            seg = ast.get_source_segment(text, node)
            defs[nm] = {"lines": [node.lineno, node.end_lineno],
                        "source_sha256": hashlib.sha256(seg.encode()).hexdigest(),
                        "ast_sha256": hashlib.sha256(ast.dump(node, include_attributes=False).encode()).hexdigest()}
    missing = sorted(set(names) - set(defs))
    assert not missing, (f, missing)
    out["files"][rel] = {"blob": blob, "file_sha256": hashlib.sha256(src).hexdigest(), "definitions": defs}
print(json.dumps(out, indent=1))
