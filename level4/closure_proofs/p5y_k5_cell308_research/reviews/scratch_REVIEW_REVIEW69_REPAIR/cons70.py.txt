def git(d, *a, inp=None, check=True, env=None):
    r = subprocess.run(["git", "-C", str(d), *CFG, *a], capture_output=True, input=inp, env=env)
    if check and r.returncode:
        raise RuntimeError(f"git {a}: {r.stderr.decode()[-400:]}")
    return r.stdout.decode().strip()


def init(d, fmt=None):
    Path(d).mkdir(parents=True, exist_ok=True)
    git(d, "init", "-q", *([f"--object-format={fmt}"] if fmt else []))
    for k, v in (("gc.auto", "0"), ("core.hooksPath", os.devnull), ("maintenance.auto", "false")):
        git(d, "config", "--local", k, v)


def commit(d, files=None, rm=()):
    for rel, data in (files or {}).items():
        p = Path(d, rel)
        p.parent.mkdir(parents=True, exist_ok=True)
        if isinstance(data, tuple):
            os.symlink(data[1], p)
        else:
            p.write_bytes(data)
    for rel in rm:
        git(d, "rm", "-q", "-r", rel)
    git(d, "add", "-A")
    git(d, "commit", "-q", "--allow-empty", "-m", "c")
    return git(d, "rev-parse", "HEAD")


def mkcommit(d, tree, parents, msg="c"):
    body = f"tree {tree}\n" + "".join(f"parent {p}\n" for p in parents)
    body += f"author t <t@t> 1700000000 +0000\ncommitter t <t@t> 1700000000 +0000\n\n{msg}\n"
    return git(d, "hash-object", "-t", "commit", "-w", "--stdin", inp=body.encode())


def raw_tree(entries):
    return b"".join(m.encode() + b" " + n.encode() + b"\0" + bytes.fromhex(o) for m, n, o in entries)


def mktree_literal(d, entries):
    return git(d, "hash-object", "-t", "tree", "-w", "--literally", "--stdin", inp=raw_tree(entries))


def valid(d, path="att/HOST.json"):
    try:
        rec, why = L.attestation_valid({"path": path, "sha256": SHA}, repo_root=str(d))
        return {"valid": rec is not None, "reasons": why}
    except Exception as e:
        return {"valid": None, "exception": f"{type(e).__name__}: {e}"}


def dircheck(d):
    try:
        return L._review_dir_is_a_directory(str(d))
    except Exception as e:
        return f"EXC {type(e).__name__}: {e}"


def fsck(d):
    r = subprocess.run(["git", "-C", str(d), *L._GIT_ISOLATION, "fsck", "--full", "--strict", "--no-dangling",
                        "--no-progress"], capture_output=True, env=L._git_env())
    err = [ln for ln in r.stderr.decode(errors="replace").splitlines() if not ln.startswith("hint:")]
    return {"rc": r.returncode, "err": err[-4:]}


def removed_later(d, bad_commit):
    """Make HEAD a child of bad_commit whose tree is the first commit's tree, then add the acceptance."""
    first = git(d, "rev-list", "--max-parents=0", "HEAD").split()[0]
    c = mkcommit(d, git(d, "rev-parse", f"{first}^{{tree}}"), [bad_commit], "rm")
    git(d, "update-ref", "HEAD", c)
    git(d, "reset", "-q", "--hard")
    return commit(d, {f"{ACC}1.md": OK})


def nested(d, leaf_entries, upto=len(TOP)):
    """A tree for TOP[:upto] holding leaf_entries at its bottom; returns the oid of the TOP[0] tree."""
    sub = mktree_literal(d, leaf_entries)
    for name in reversed(TOP[1:upto]):
        sub = mktree_literal(d, [("40000", name, sub)])
    return sub

def long_history(w, n):
    d = w / "r"; init(d)
    stream = [b"blob\nmark :1\ndata %d\n" % len(ATT) + ATT + b"\n", b"blob\nmark :2\ndata %d\n" % len(OK) + OK + b"\n"]
    for i in range(n):
        s = f"commit refs/heads/main\ncommitter t <t@t> {1700000000 + i} +0000\ndata 1\nx\n"
        if i == 0:
            s += "M 100644 :1 att/HOST.json\n"
        else:
            s += f"M 100644 inline {RD}/n{i % 5}.md\ndata {len(str(i)) + 1}\n{i}\n"
        if i == n - 1:
            s += f"M 100644 :2 {ACC}1.md\n"
        stream.append(s.encode())
    subprocess.run(["git", "-C", str(d), "fast-import", "--quiet"], input=b"".join(stream), check=True)
    git(d, "reset", "-q", "--hard", "main")
    t0 = time.time(); dc = dircheck(d); t1 = time.time()
    return {"commits": n, "distinct_trees_every_depth": True, "dir_check": dc, "dir_check_s": round(t1 - t0, 2)}


@kind("long_history_30000_deep_changes")
def _(w):
    return long_history(w, 30000)


@kind("long_history_150000_deep_changes")
def _(w):
    return long_history(w, 150000)


for name, fn in KINDS.items():
    if ONLY and name not in ONLY:
        continue
    try:
        with tempfile.TemporaryDirectory(dir=TMP) as w:
            results[name] = fn(Path(w))
    except Exception:
        results[name] = {"probe_error": traceback.format_exc()[-1500:]}
    if not ONLY:
        print(name, json.dumps(results[name].get("result", results[name].get("dir_check", "..."))
                               if isinstance(results[name], dict) else results[name])[:300], flush=True)
Path(OUT).write_text(json.dumps(results, indent=1, sort_keys=True) + "\n")
if ONLY:
    print(json.dumps({k: results[k] for k in ONLY}))
