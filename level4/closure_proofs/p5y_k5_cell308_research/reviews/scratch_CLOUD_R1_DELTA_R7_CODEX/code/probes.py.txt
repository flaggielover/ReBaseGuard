def check(name, fn):
    try:
        details = fn()
        CASES.append({'name': name, 'pass': True, 'details': details})
    except Exception as exc:
        CASES.append({'name': name, 'pass': False, 'error': type(exc).__name__ + ': ' + str(exc)})

def require(value, text):
    if not value:
        raise AssertionError(text)

def git(repo, *args):
    p = subprocess.run(['git', '-C', str(repo), '-c', 'user.name=Independent synthetic fixture',
                        '-c', 'user.email=review65@example.invalid', '-c', 'commit.gpgSign=false', *args],
                       capture_output=True, check=True, text=True)
    return p.stdout

def launch(inv='independent-live'):
    return {'systemd_pid1': True, 'unit': L.CAMPAIGN_UNIT_PREFIX+'synthetic.service',
            'invocation_id': inv, 'systemctl': {'MainPID': os.getpid(), 'InvocationID': inv}, 'pass': True}

def fixture(repo, *, path='fixture/host.json', commit=True, review=True, **changes):
    rec = {'schema': L.ATTESTATION_SCHEMA, 'reviewed': True, 'accepted': True, 'machine': HW,
           'acceptance_record': 'fixture/review.txt', 'root_cgroup_is_real': True,
           'provider_image_updates': L.NO_IMAGE_CHANGE}
    rec.update(changes)
    raw = (json.dumps(rec, ensure_ascii=True, sort_keys=True)+'\n').encode()
    digest = hashlib.sha256(raw).hexdigest()
    dest = repo/path; dest.parent.mkdir(parents=True, exist_ok=True); dest.write_bytes(raw)
    (repo/'fixture/review.txt').write_text(L.ACCEPTANCE_LINE+digest+'\n' if review else 'refused\n')
    if commit:
        git(repo, 'add', 'fixture'); git(repo, 'commit', '-q', '--allow-empty', '-m', 'Synthetic infrastructure only')
    return {'path': path, 'sha256': digest}

def invariants(out, expected_verified, expected_root):
    require(out['attestation']['verified'] is expected_verified, 'wrong attestation verified flag')
    require(out['attestation']['root_cgroup_is_real'] is expected_root, 'wrong strict root flag')
    require(out['raw_memory']['readings']['root_attested'] is expected_root, 'wrong helper root contract')
    require(out['development_only'] is True and out['operational_ready'] is False, 'authority boundary changed')
    require(out['conditions']['A5'] == 'UNMET_FIXED_SCIENCE_IMPORT_LIST_AND_PIN_POINT', 'A5 changed')
    require(out['raw_memory']['classification'] == 'RAW_LINUX_READINGS', 'classification changed')
    require(not ({'effective_limit', 'cgroup_min_headroom'} & set(out['raw_memory']['readings'])), 'allowance leaked')
    pf = out['preflight']
    require(pf['pass'] is False, 'preflight pass escaped open conditions')
    require(pf['gates']['no_other_campaign_job'] == L.UNKNOWN, 'detector became decided')
    require(pf['gates']['host_lifetime_attested'] == L.OPEN_DECISION, 'host authority became decided')
    require(bool(pf['open_decisions']), 'empty open conditions')
    return {'verified': expected_verified, 'root': expected_root, 'reasons': out['attestation']['reasons'],
            'preflight_pass': pf['pass'], 'detector': pf['gates']['no_other_campaign_job']}
