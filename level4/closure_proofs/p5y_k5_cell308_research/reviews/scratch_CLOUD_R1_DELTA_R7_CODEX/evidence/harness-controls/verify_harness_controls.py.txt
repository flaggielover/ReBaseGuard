def run_suite_control(label, side_effect=None, completion=None):
    supplied_env = {'LX_PLATFORM_DIR': 'synthetic-fixture', 'PYTHONDONTWRITEBYTECODE': '0'}
    before = dict(supplied_env)
    with mock.patch.object(h.subprocess, 'run', side_effect=side_effect, return_value=completion) as runner, \
         mock.patch.object(h.time, 'monotonic', side_effect=[10.0, 11.0]), \
         mock.patch.object(h.sys, 'executable', 'SYNTHETIC_PYTHON'):
        record = h.run_suite(supplied_env, timeout_s=7)
    assert supplied_env == before
    runner.assert_called_once()
    args, kwargs = runner.call_args
    assert args[0] == ['SYNTHETIC_PYTHON', '-B', '-W', 'error::ResourceWarning', '-m',
                       'unittest', 'discover', '-s', str(h.HERE), '-v']
    assert kwargs == {'cwd': h.HERE.parent, 'env': dict(before, PYTHONDONTWRITEBYTECODE='1'),
                      'capture_output': True, 'text': True, 'timeout': 7}
    assert record['elapsed_s'] == 1.0
    run_suite_controls.append({'control': label, 'subprocess_was_mocked': True, 'command': args[0],
                               'environment_controls': record['environment_controls'],
                               'invalid_reasons': record['invalid_reasons'], 'stdout': record['stdout'],
                               'stderr': record['stderr'], 'passed': True})
    return record

class FakePath:
    def __init__(self, value):
        self.value = PurePosixPath(value.value if isinstance(value, FakePath) else value)
        assert self.value.is_relative_to(fixture_root)

    def __truediv__(self, value):
        return FakePath(self.value / value)

    @property
    def parent(self):
        return FakePath(self.value.parent)

    def resolve(self):
        return self

    def relative_to(self, other):
        return self.value.relative_to(other.value)

    def read_bytes(self):
        key = str(self.value.relative_to(fixture_root))
        fixture_reads.append(key)
        return fixture_bytes[key]

class FixedDatetime(datetime.datetime):
    @classmethod
    def now(cls, tz=None):
        return cls(2026, 10, 15, 0, 0, 0, tzinfo=tz)


def main_control(label, *, baseline=PASS, outcomes=None, first_not_applied=False,
                 after_changed=False, registry=None, source_hash_mismatch=False, expected_exit=1,
                 expected_exception=None, write_output=False):
    run_calls, apply_calls, temp_calls = [], [], []
    inherited_fixture = {'LX_PLATFORM_DIR': 'INHERITED_SYNTHETIC_CONTROL', 'PYTHONDONTWRITEBYTECODE': '0'}
    before = dict(BEFORE_HASHES)
    if source_hash_mismatch:
        before['lx_platform.py'] = '0' * 64
    after = dict(before)
    if after_changed:
        after['lx_inventory.py'] = '1' * 64
    registry = ORIGINAL_REGISTRY if registry is None else registry
    outcomes = {} if outcomes is None else outcomes

    def fake_run_suite(env):
        env = dict(env)
        run_calls.append(env)
        assert env['PYTHONDONTWRITEBYTECODE'] == '1'
        if len(run_calls) == 1:
            assert 'LX_PLATFORM_DIR' not in env
            chosen = baseline
        else:
            assert Path(env['LX_PLATFORM_DIR']).is_relative_to(temp_root)
            assert set(env) == {'LX_PLATFORM_DIR', 'PYTHONDONTWRITEBYTECODE'}
            index = len(run_calls) - 2 + (1 if first_not_applied else 0)
            chosen = outcomes.get(index, ASSERT)
        result = dict(chosen)
        result['environment_controls'] = {'LX_PLATFORM_DIR': env.get('LX_PLATFORM_DIR'),
                                          'PYTHONDONTWRITEBYTECODE': env.get('PYTHONDONTWRITEBYTECODE')}
        return result

    def fake_apply(src, edits):
        assert src == SYNTHETIC_SOURCE
        index = len(apply_calls)
        apply_calls.append(index)
        if first_not_applied and index == 0:
            return None, [0], 'ANCHOR_NOT_UNIQUE'
        return SYNTHETIC_MUTATED, [1] * len(edits), None

    def bounded_tempdir():
        result = real_tempdir(prefix='synthetic-', dir=str(temp_root))
        temp_calls.append(result.name)
        assert Path(result.name).is_relative_to(temp_root)
        return result

    printed = io.StringIO()
    output_path = ROOT / f'main-{label}.json'
    caught = None
    with contextlib.ExitStack() as stack:
        patches = [mock.patch.object(h, 'input_hashes', side_effect=[before, after]),
                   mock.patch.object(h, 'SRC', SimpleNamespace(read_text=lambda: SYNTHETIC_SOURCE)),
                   mock.patch.object(h, 'run_suite', side_effect=fake_run_suite),
                   mock.patch.object(h, 'apply_mutation', side_effect=fake_apply),
                   mock.patch.object(h, 'MUTANTS', registry),
                   mock.patch.object(h.os, 'environ', inherited_fixture),
                   mock.patch.object(h.os.path, 'realpath', return_value='SYNTHETIC_REAL_PYTHON'),
                   mock.patch.object(h.sys, 'executable', 'SYNTHETIC_PYTHON'),
                   mock.patch.object(h.sys, 'version', 'SYNTHETIC_PYTHON_VERSION'),
                   mock.patch.object(h.platform, 'python_implementation', return_value='SYNTHETIC_IMPLEMENTATION'),
                   mock.patch.object(h.platform, 'release', return_value='SYNTHETIC_KERNEL'),
                   mock.patch.object(h.platform, 'machine', return_value='SYNTHETIC_MACHINE'),
                   mock.patch.object(h.platform, 'system', return_value='SYNTHETIC_SYSTEM'),
                   mock.patch.object(h.datetime, 'datetime', FixedDatetime),
                   mock.patch.object(h.tempfile, 'TemporaryDirectory', side_effect=bounded_tempdir)]
        for patch in patches:
            stack.enter_context(patch)
        stack.enter_context(contextlib.redirect_stdout(printed))
        try:
            exit_value = h.main(str(output_path) if write_output else None)
        except Exception as exc:
            caught = exc
    assert h.os.environ is original_environ_object
    assert inherited_fixture == {'LX_PLATFORM_DIR': 'INHERITED_SYNTHETIC_CONTROL', 'PYTHONDONTWRITEBYTECODE': '0'}
    assert not any(Path(name).exists() for name in temp_calls)
    if expected_exception:
        assert isinstance(caught, expected_exception), (label, caught)
        assert printed.getvalue() == ''
        record = None
        save(output_path.name, {'synthetic_control': label, 'expected_exception': type(caught).__name__,
                                'message': str(caught)})
    else:
        assert caught is None, (label, caught)
        assert exit_value == expected_exit, (label, exit_value)
        record = json.loads(printed.getvalue())
        assert record['new_target_evaluations'] == 0
        assert record['input_sha256'] == before and record['input_sha256_after'] == after
        assert record['expected_mutants'] == 80
        if write_output:
            assert output_path.read_text() == printed.getvalue()
        else:
            output_path.write_text(printed.getvalue())
    main_controls.append({'control': label, 'expected_exit': expected_exit if not expected_exception else None,
                          'exception': type(caught).__name__ if caught else None,
                          'suite_calls': len(run_calls), 'mocked_apply_calls': len(apply_calls),
                          'bounded_synthetic_tempdirs': len(temp_calls),
                          'status_counts': record['status_counts'] if record else None,
                          'inputs_unchanged': record['inputs_unchanged'] if record else None,
                          'record_artifact': output_path.name, 'environment_restored': True, 'passed': True})
    return record, run_calls, apply_calls, temp_calls
