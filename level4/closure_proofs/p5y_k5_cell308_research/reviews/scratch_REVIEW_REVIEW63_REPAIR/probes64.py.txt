class Repo:
    def __init__(self, d):
        self.d, self.t = d, 1700000000
        self.git("init", "-q")
        for k, v in (("gc.auto", "0"), ("core.hooksPath", os.devnull), ("maintenance.auto", "false")):
            self.git("config", "--local", k, v)

    def git(self, *a, check=True, date=None, inp=None):
        self.t += 60
        ts = str(date or self.t)
        env = dict(os.environ, GIT_AUTHOR_DATE=f"{ts} +0000", GIT_COMMITTER_DATE=f"{ts} +0000")
        return subprocess.run(["git", "-C", self.d, *CFG, *a], check=check, capture_output=True, env=env, input=inp)

    def commit(self, files=None, date=None, msg="c"):
        for rel, data in (files or {}).items():
            p = Path(self.d, rel)
            p.parent.mkdir(parents=True, exist_ok=True)
            if isinstance(data, tuple) and data[0] == "symlink":
                os.symlink(data[1], p)
            else:
                p.write_bytes(data)
        self.git("add", "-A")
        self.git("commit", "-q", "--allow-empty", "-m", msg, date=date)

    def valid(self, sha=SHA, path="att/HOST.json"):
        try:
            rec, why = L.attestation_valid({"path": path, "sha256": sha}, repo_root=self.d)
            return {"valid": rec is not None, "reasons": why}
        except Exception as e:  # a crash path is a finding
            return {"valid": None, "exception": f"{type(e).__name__}: {e}", "tb": traceback.format_exc()[-600:]}


def probe(name, steps, **kw):
    with tempfile.TemporaryDirectory(dir=TMP) as d:
        r = Repo(d)
        try:
            extra = steps(r) or {}
        except subprocess.CalledProcessError as e:
            results[name] = {"setup_error": (e.stderr or b"").decode(errors="replace")[-400:]}
            return
        res = r.valid(**kw)
        res.update(extra)
        results[name] = res

# ---------------------------------------------------------------- unit_enabled (file mode and systemd mode)
def ue(name, files, unit, roots=("hi", "lo"), pid1=False, act=None, st=None):
    with tempfile.TemporaryDirectory(dir=TMP) as d:
        for rel, v in files.items():
            p = Path(d, rel)
            p.parent.mkdir(parents=True, exist_ok=True)
            if isinstance(v, tuple):
                os.symlink(v[1], p)
            elif v is None:
                p.mkdir(exist_ok=True)
            else:
                p.write_text(v)
        for rt in roots:
            Path(d, rt).mkdir(exist_ok=True)
        saved = (L.systemd_is_pid1, L._systemctl_is_active, L._systemctl_is_enabled)
        L.systemd_is_pid1 = lambda *a, **k: pid1
        L._systemctl_is_active = lambda u: act
        L._systemctl_is_enabled = lambda u: st
        try:
            results[name] = L.unit_enabled(unit, tuple(str(Path(d, rt)) for rt in roots))
        except Exception as e:
            results[name] = f"EXCEPTION {type(e).__name__}: {e}"
        finally:
            L.systemd_is_pid1, L._systemctl_is_active, L._systemctl_is_enabled = saved
