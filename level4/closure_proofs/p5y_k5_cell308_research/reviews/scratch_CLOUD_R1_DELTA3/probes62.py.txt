@contextlib.contextmanager
def patched(**attrs):
    saved = {k: getattr(L, k) for k in attrs}
    for k, v in attrs.items():
        setattr(L, k, v)
    try:
        yield
    finally:
        for k, v in saved.items():
            setattr(L, k, v)


def mk(root, files):
    for rel, txt in files.items():
        p = Path(root, rel); p.parent.mkdir(parents=True, exist_ok=True)
        p.mkdir(parents=True, exist_ok=True) if txt is None else p.write_text(txt)


def commit(d, files, msg="c"):
    for rel, data in files.items():
        Path(d, rel).parent.mkdir(parents=True, exist_ok=True)
        if data is None:
            Path(d, rel).unlink()
        else:
            Path(d, rel).write_bytes(data)
    subprocess.run(["git", "-C", d, "add", "-A"], check=True)
    subprocess.run(["git", "-C", d, "-c", "user.name=t", "-c", "user.email=t@t", "-c", "commit.gpgsign=false",
                    "commit", "-q", "-m", msg], check=True)
