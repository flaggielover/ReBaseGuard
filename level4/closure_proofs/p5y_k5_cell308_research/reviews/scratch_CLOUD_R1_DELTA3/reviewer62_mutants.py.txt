written to a temporary directory under this scratch dir; the UNCHANGED repository suite is run against it via
LX_PLATFORM_DIR with -B. Nothing in the repository is written. Usage: python3 -B reviewer62_mutants.py"""

SRC = HERE / "base" / "lx_platform.py"
TESTS = Path("/home/user/ReBaseGuard/level4/closure_proofs/p5y_k5_cell308_research/streams/CLOUD_R1_LINUX_DEV/tests")
MUTANTS = [

def run_suite(d):
    env = dict(os.environ, LX_PLATFORM_DIR=d, PYTHONDONTWRITEBYTECODE="1")
    p = subprocess.run([sys.executable, "-B", "-m", "unittest", "discover", "-s", str(TESTS)], cwd=str(TESTS.parent),
                       env=env, capture_output=True, text=True, timeout=900)
    fails = sorted({ln.split()[1] for ln in p.stderr.splitlines() if ln.startswith(("FAIL:", "ERROR:"))})
    return p.returncode, (p.stderr.strip().splitlines() or [""])[-1], fails


src = SRC.read_text()
rows = []
scratch = HERE / "mut_tmp"; scratch.mkdir(exist_ok=True)
for mid, what, edits in MUTANTS:
    m = src
    for old, new in edits:
        assert m.count(old) == 1, (mid, old)
        m = m.replace(old, new)
    diff = "".join(difflib.unified_diff(src.splitlines(True), m.splitlines(True), n=0))
    diff = "".join(l for l in diff.splitlines(True) if not l.startswith(("---", "+++")))
    with tempfile.TemporaryDirectory(dir=scratch) as d:
        Path(d, "lx_platform.py").write_text(m)
        rc, tail, fails = run_suite(d)
    rows.append({"id": mid, "property": what, "status": "KILLED" if rc else "SURVIVED", "tail": tail, "killed_by": fails,
                 "diff": diff})
with tempfile.TemporaryDirectory(dir=scratch) as d:
    Path(d, "lx_platform.py").write_text(src)
    rc, tail, _ = run_suite(d)
print(json.dumps({"base_copy_suite": {"rc": rc, "tail": tail}, "mutants": rows}, indent=1))
