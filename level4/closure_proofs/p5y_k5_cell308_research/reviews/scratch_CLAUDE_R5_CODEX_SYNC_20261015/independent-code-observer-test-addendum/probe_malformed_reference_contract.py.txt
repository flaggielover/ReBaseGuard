ROWS=[]

    t=module.TestVerifiedObservation('test_invalid_reference_paths_are_refused_without_exceptions')
    t.setUp()

        path_cases=[None,True,7,b'bytes-path',[],{},'','missing.json','\0','prefix\0suffix','\ud800','prefix\ud800suffix','\udfff']

            assert result['operational_ready'] is False
            ROWS.append({'case':'malformed_path '+repr(value),'verified':result['attestation']['verified'],'has_reasons':bool(result['attestation']['reasons']),'operational_ready':result['operational_ready']})

            result=t.assert_unverified(ref)

        positive=V.observe(t.fixture(path='fixtures/hôte.json'))

        ROWS.append({'case':'valid_committed_UTF8_path_positive','verified':True,'operational_ready':positive['operational_ready']})
        # Exercise the narrow handler itself; no product source or mutant is changed.
        for exception in (ValueError('synthetic embedded NUL'),UnicodeError('synthetic malformed encoding'),UnicodeEncodeError('utf-8','\ud800',0,1,'synthetic surrogate'),TypeError('unrelated unexpected failure'),RuntimeError('unrelated unexpected failure'),AssertionError('existing assertion must remain a failure')):
            try:
                with mock.patch.object(t,'assert_unverified',side_effect=exception):
                    t.test_invalid_reference_paths_are_refused_without_exceptions()
            except Exception as caught:
                if isinstance(exception,(ValueError,UnicodeError)):
                    assert type(caught) is AssertionError
                    assert str(caught).startswith('Malformed reference must be refused without leaking ')
                    expectation='explicit unittest failure for advertised malformed-reference contract'
                else:
                    assert caught is exception
                    expectation='unrelated exception unchanged; no blanket handler'
                ROWS.append({'case':'handler '+type(exception).__name__,'raised':type(caught).__name__,'expectation':expectation,'matched_expected':True})
            else:raise AssertionError('no exception expected to be swallowed')
    finally:t.doCleanups()
