def meta(text,rc,exception_type='builtins.AssertionError',exact=True):
    count=re.findall(r'^Ran (\d+) tests? in',text,re.M)
    events=[]
    for kind,header in [('failure','FAIL'),('error','ERROR')]:
        for name in re.findall('^'+header+r': (.+)$',text,re.M):
            m=re.match(r'([^ ]+) \(([^)]+)\)',name)
            parent=m.group(2) if m else IDS[0]
            events.append({'kind':kind,'test_id':parent,'parent_test_id':parent,'description':name,'exception_type':exception_type if kind=='failure' else 'builtins.RuntimeError','is_exact_assertion_error':exact if kind=='failure' else False,'traceback_frames':[dict(FRAME)]})
    return {'schema':'CELL308_CONTROLLED_UNITTEST_RESULT_V1','ran':int(count[0]) if count else 2,'successful':rc==0,'executed_test_ids':IDS.copy(),'events':events,'skipped':0,'expected_failures':0,'unexpected_successes':0}
