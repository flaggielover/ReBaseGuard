    return {'clocks':{'monotonic_ns':ns,'boottime_ns':ns,'realtime_ns':ns},'suspend_success':7,'boot_id':'malformed-boot','power':L.AC}

    result=L.assess(snap(0),snap(30),[snap(10),snap(20)],[])
assert result=={'status':'AMBIGUOUS','clean':False,'reasons':['AMBIGUOUS: a boot_id reading is absent']},result
assert DENIED==[],DENIED
