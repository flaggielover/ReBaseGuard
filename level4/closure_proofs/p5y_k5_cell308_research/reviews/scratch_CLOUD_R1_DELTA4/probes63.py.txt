@contextlib.contextmanager
def patched(**attrs):
    saved = {k: getattr(L, k) for k in attrs}
    for k, v in attrs.items():
        setattr(L, k, v)
    try:
        yield
    finally:
        for k, v in saved.items():
            setattr(L, k, v)


def mk(root, files):

        p.mkdir(parents=True, exist_ok=True) if txt is None else p.write_text(txt)

def git(d, *a, date=None, check=True):
    env = dict(GENV)
    if date:
        env.update(GIT_AUTHOR_DATE=date, GIT_COMMITTER_DATE=date)
    return subprocess.run(["git", "-C", d, "-c", "commit.gpgsign=false", *a], check=check, env=env,
                          capture_output=True)


def commit(d, files, msg="c", date=None):
    for rel, data in files.items():
        Path(d, rel).parent.mkdir(parents=True, exist_ok=True)
        if data is None:
            Path(d, rel).unlink()
        else:
            Path(d, rel).write_bytes(data)
    git(d, "add", "-A")
    git(d, "commit", "-q", "--allow-empty", "-m", msg, date=date)


def init(d):
