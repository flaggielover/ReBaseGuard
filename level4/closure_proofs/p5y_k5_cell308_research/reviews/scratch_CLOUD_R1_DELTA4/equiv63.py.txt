"""Brief 63: for each reviewer mutant W01-W08, an input on which the mutant and revision 5 (copy in ./base) give
different outcomes. Synthetic only (temporary dirs / git repos under ./tmp). Usage: python3 -B equiv63.py"""
import hashlib, importlib.util, json, os, re, subprocess, sys, tempfile
from pathlib import Path
HERE = Path(__file__).resolve().parent
TMP = HERE / "tmp"; TMP.mkdir(exist_ok=True)
src = (HERE / "base" / "lx_platform.py").read_text()
ns = {}
exec(re.sub(r"^.*?(MUTANTS = \[)", r"\1", (HERE / "reviewer63_mutants.py").read_text().split("\n\ndef run_suite")[0],
            flags=re.S), ns)
HW = {"machine_id_sha256": "a" * 64, "product_uuid_sha256": "b" * 64, "hardware_bound": True}
BOOT = "0f1e2d3c-4b5a-6978-8796-a5b4c3d2e1f0"


def load(text, name):
    d = tempfile.mkdtemp(dir=TMP); p = Path(d, "lx_platform.py"); p.write_text(text)
    spec = importlib.util.spec_from_file_location(name, p); m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m); return m


def commit(d, files, date):
    for rel, data in files.items():
        Path(d, rel).parent.mkdir(parents=True, exist_ok=True); Path(d, rel).write_bytes(data)
    env = dict(os.environ, GIT_AUTHOR_NAME="t", GIT_AUTHOR_EMAIL="t@t", GIT_COMMITTER_NAME="t", GIT_COMMITTER_EMAIL="t@t",
               GIT_AUTHOR_DATE=date, GIT_COMMITTER_DATE=date)
    subprocess.run(["git", "-C", d, "add", "-A"], check=True, env=env)
    subprocess.run(["git", "-C", d, "-c", "commit.gpgsign=false", "commit", "-q", "-m", "c"], check=True, env=env)


def scenario(L, mid):
    if mid in ("W01", "W06"):
        a = json.dumps({"schema": L.ATTESTATION_SCHEMA, "reviewed": True, "accepted": True, "machine": HW}).encode()
        sha = hashlib.sha256(a).hexdigest()
        acc = f"{L.ACCEPTANCE_REVIEW_DIR}/{L.ACCEPTANCE_REVIEW_PREFIX}1.md"
        line = f"x\n{L.ACCEPTANCE_LINE}{sha}\n{L.ACCEPTANCE_VERDICT_LINE}\n".encode()
        with tempfile.TemporaryDirectory(dir=TMP) as d:
            subprocess.run(["git", "init", "-q", d], check=True)
            if mid == "W01":      # acceptance written BEFORE the attestation, quoted again later
                commit(d, {acc: line}, "2026-10-01T00:00:00Z"); commit(d, {"att/H.json": a}, "2026-10-02T00:00:00Z")
                commit(d, {"notes/q.md": f"{L.ACCEPTANCE_LINE}{sha}\n".encode()}, "2026-10-03T00:00:00Z")
            else:                 # a later review revokes the acceptance
                commit(d, {"att/H.json": a}, "2026-10-01T00:00:00Z"); commit(d, {acc: line}, "2026-10-02T00:00:00Z")
                commit(d, {f"{L.ACCEPTANCE_REVIEW_DIR}/REVIEW_LATER.md": f"Acceptance of {sha} revoked.\n".encode()},
                       "2026-10-03T00:00:00Z")
            L.machine_identity = lambda *a, **k: HW
            return L.attestation_valid({"path": "att/H.json", "sha256": sha}, repo_root=d)[0] is not None
    if mid in ("W02", "W03"):
        rec = {"pid": 4242, "starttime_ticks": 777, "boot_id": BOOT, "cmdline_sha256": "c" * 64, "pidns": 1, "timens": 2}
        rec = dict(rec, cmdline_sha256="C" * 64) if mid == "W02" else dict(rec, boot_id=BOOT.upper())
        st = L.proc_stat(4242, "4242 (x) S 1 1 1 0 -1 0 0 0 0 0 10 5 0 0 20 0 1 0 777 0 0\n")
        return L.identity_state(rec, current_boot=BOOT, exists=lambda p: True, stat=lambda p: st,
                                cmd=lambda p: "c" * 64, current_ns={"pidns": 1, "timens": 2})
    L.systemd_is_pid1 = lambda *a, **k: True
    L._systemctl_is_active = lambda u: "inactive"
    L._systemctl_is_enabled = (lambda u: None) if mid == "W08" else (lambda u: "not-found")
    with tempfile.TemporaryDirectory(dir=TMP) as d:
        r1, r2 = Path(d, "r1"), Path(d, "r2")
        if mid == "W04":
            r1.mkdir(); os.symlink("/nonexistent/x.timer", r1 / "x.timer"); return L.unit_enabled("x.timer", (str(r1),))
        if mid == "W05":
            r1.write_text("not a dir"); return L.unit_enabled("x.timer", (str(r1),))
        if mid == "W07":
            r1.mkdir(); r2.mkdir(); (r1 / "x.timer").write_text(""); return L.unit_enabled("x.timer", (str(r1), str(r2)))
        if mid == "W08":
            return L.unit_enabled("x.timer", ())


res = {}
base_hash = hashlib.sha256(src.encode()).hexdigest()
for mid, what, edits in ns["MUTANTS"]:
    m = src
    for old, new in edits:
        assert m.count(old) == 1, (mid, old); m = m.replace(old, new)
    res[mid] = {"rev5": scenario(load(src, f"base_{mid}"), mid), "mutant": scenario(load(m, f"mut_{mid}"), mid)}
print(json.dumps({"base_sha256": base_hash, "results": res}, indent=1, default=str))
