def git(d, *a, inp=None, check=True):
    r = subprocess.run(["git", "-C", str(d), *CFG, *a], capture_output=True, input=inp)
    if check and r.returncode:
        raise RuntimeError(f"git {a}: {r.stderr.decode()[-400:]}")
    return r.stdout.decode().strip()


def init(d):
    Path(d).mkdir(parents=True, exist_ok=True)
    git(d, "init", "-q")
    for k, v in (("gc.auto", "0"), ("core.hooksPath", os.devnull), ("maintenance.auto", "false")):
        git(d, "config", "--local", k, v)


def commit(d, files=None, rm=()):
    for rel, data in (files or {}).items():
        p = Path(d, rel)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(data)
    for rel in rm:
        git(d, "rm", "-q", rel)
    git(d, "add", "-A")
    git(d, "commit", "-q", "--allow-empty", "-m", "c")
    return git(d, "rev-parse", "HEAD")


def valid(d, path="att/HOST.json", sha=SHA):
    try:
        rec, why = L.attestation_valid({"path": path, "sha256": sha}, repo_root=str(d))
        return {"valid": rec is not None, "reasons": why}
    except Exception as e:
        return {"valid": None, "exception": f"{type(e).__name__}: {e}"}


def hist(d):
    try:
        return L._history_complete(str(d))
    except Exception as e:
        return f"EXC {type(e).__name__}: {e}"


def raw(d, *a):
    r = subprocess.run(["git", "-C", str(d), *L._GIT_ISOLATION, *a], capture_output=True, env=L._git_env())
    return {"rc": r.returncode, "out": r.stdout.decode(errors="replace")[-600:], "err": r.stderr.decode(errors="replace")[-300:]}


def mkcommit(d, tree, parents, msg="c", extra_headers=""):
    body = f"tree {tree}\n" + "".join(f"parent {p}\n" for p in parents)
    body += "author t <t@t> 1700000000 +0000\ncommitter t <t@t> 1700000000 +0000\n" + extra_headers + f"\n{msg}\n"
    return git(d, "hash-object", "-t", "commit", "-w", "--stdin", inp=body.encode())


def mktree_literal(d, entries):
    """entries: list of (mode, name, oid) already sorted by the caller; written with --literally (no checks)."""
    data = b"".join(m.encode() + b" " + n.encode() + b"\0" + bytes.fromhex(o) for m, n, o in entries)
    return git(d, "hash-object", "-t", "tree", "-w", "--literally", "--stdin", inp=data)


def probe(name):
    def deco(fn):
        try:
            with tempfile.TemporaryDirectory(dir=TMP) as w:
                results[name] = fn(Path(w))
        except Exception:
            results[name] = {"probe_error": traceback.format_exc()[-1500:]}
        return fn
    return deco

def signed_child(d, parent, files):
    """A commit with `files` on top of `parent` (via the index), carrying a (fake) gpgsig header."""
    for rel, data in files.items():
        p = Path(d, rel)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_bytes(data)
    git(d, "add", "-A")
    tree = git(d, "write-tree")
    c = mkcommit(d, tree, [parent], extra_headers=GPG_SIG)
    git(d, "update-ref", "HEAD", c)
    return c




def wt_read(d, commit_id, rel):
    """What a checkout of commit_id shows at rel (follows symlinks like a reader would)."""
    wt = Path(str(d) + "_wt_" + commit_id[:8])
    git(d, "worktree", "add", "-q", "--detach", str(wt), commit_id)
    try:
        p = wt / rel
        return {"exists": p.exists(), "is_symlink": p.is_symlink(),
                "text": p.read_text()[:200] if p.exists() and p.is_file() else None}
    finally:
        git(d, "worktree", "remove", "--force", str(wt))


def fsck(d):
    a = raw(d, "fsck", "--full", "--strict", "--no-dangling", "--no-progress")
    return {"rc": a["rc"], "err_tail": a["err"][-200:]}

def batch(d, names):
    r = subprocess.run(["git", "-C", str(d), *L._GIT_ISOLATION, "cat-file", "--batch-check"],
                       input=("\n".join(names) + "\n").encode(), capture_output=True, env=L._git_env())

def index_gitlink(d, path, oid):
    git(d, "update-index", "--add", "--cacheinfo", f"160000,{oid},{path}")


def put_entry(d, parent_commit, path, mode, oid, msg="odd"):
    """A commit on parent_commit whose tree has `path` replaced by a (mode, oid) entry, built via the index."""
    git(d, "read-tree", parent_commit)
    git(d, "update-index", "--force-remove", path, check=False)
    git(d, "update-index", "--add", "--cacheinfo", f"{mode},{oid},{path}")
    tree = git(d, "write-tree")
    c = git(d, "commit-tree", tree, "-p", parent_commit, "-m", msg)
    git(d, "update-ref", "HEAD", c)
    git(d, "read-tree", c)
    return c

@probe("long_history_timing")
def _(w):
    import time
    d = w / "r"; init(d)
    n = 30000
    stream = [b"blob\nmark :1\ndata %d\n" % len(ATT) + ATT + b"\n"]
    stream.append(b"blob\nmark :2\ndata %d\n" % len(OK) + OK + b"\n")
    for i in range(n):
        s = f"commit refs/heads/main\ncommitter t <t@t> {1700000000 + i} +0000\ndata 1\nx\n"
        if i == 0:
            s += "M 100644 :1 att/HOST.json\n"
        else:
            s += f"M 100644 inline f{i % 7}.txt\ndata {len(str(i)) + 1}\n{i}\n"
        if i == n - 1:
            s += f"M 100644 :2 {ACC}1.md\n"
        stream.append(s.encode())
    subprocess.run(["git", "-C", str(d), "fast-import", "--quiet"], input=b"".join(stream), check=True)
    git(d, "reset", "-q", "--hard", "main")
    t0 = time.time(); dc = dircheck(d); t1 = time.time(); v = valid(d); t2 = time.time()
    return {"commits": n, "dir_check": dc, "dir_check_s": round(t1 - t0, 2), "attestation_valid_s": round(t2 - t1, 2),
            "result": v}
