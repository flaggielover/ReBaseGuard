def outcome(moddir, script, kind):
    if kind is None:
        return None
    with tempfile.TemporaryDirectory(dir=TMP) as w:
        p = subprocess.run([sys.executable, "-B", "-I", script, moddir, w, str(Path(w, "o.json")), kind],
                           capture_output=True, text=True, timeout=900)
    try:
        r = json.loads(p.stdout.strip().splitlines()[-1])[kind]
        if kind == "tree_entry_edges":
            return {"batch_newline_in_name": r.get("batch_newline_in_name")}
        return {"dir_check": r.get("dir_check"), "result": r.get("result"),
                **({"tree_entry_direct": r["tree_entry_direct"]} if "tree_entry_direct" in r else {})}
    except Exception:
        return {"child_error": p.stderr[-800:]}


src = H.SRC.read_text()
env = dict(os.environ)
env.pop("LX_PLATFORM_DIR", None)
env.update(PYTHONDONTWRITEBYTECODE="1", PYTHONWARNINGS="error", PYTHONHASHSEED="0")
base = H.run_suite(env)
out = {"source_sha256": hashlib.sha256(src.encode()).hexdigest(), "baseline_ran": base["ran"],
       "baseline_pass": H.suite_passed(base), "expected_test_count": H.expected_test_count(),
       "registry_size": len(H.validate_registry()), "registered": [], "mutants": []}
print("baseline", base["ran"], H.suite_passed(base), flush=True)


def run(mid, edits, witnesses, script=None, kind=None, based=None):
    mutated, counts, err = H.apply_mutation(src, edits)
    row = {"id": mid, "witnesses": witnesses, "anchor_counts": counts, "edits": [{"old": o, "new": n} for o, n in edits]}
    if err:
        row.update(status="NOT_APPLIED", reasons=[err])
        return row
    with tempfile.TemporaryDirectory(dir=TMP) as d:
        Path(d, "lx_platform.py").write_text(mutated)
        suite = H.run_suite(dict(env, LX_PLATFORM_DIR=d))
        if based and kind:
            row["outcome_base"] = outcome(based, script, kind)
            row["outcome_mutant"] = outcome(d, script, kind)
    status, reasons = H.classify_mutant(suite, base, witnesses)
    failed = sorted({n.split(" ", 1)[0] for n in suite["assertion_failure_names"]})
    row.update(status=status, reasons=reasons, ran=suite["ran"], returncode=suite["returncode"], failed_methods=failed,
               failure_names=sorted(suite["assertion_failure_names"]), error_names=suite["error_names"],
               only_own_witnesses=set(failed) <= set(witnesses),
               mutated_sha256=hashlib.sha256(mutated.encode()).hexdigest())
    return row


def short(names):
    return [n.split(" ", 1)[0] + (" " + n.rsplit(") ", 1)[-1] if n.endswith(")") and ") (" in n else "") for n in names]


reg = {mid: (what, edits) for mid, what, edits in H.MUTANTS}

              flush=True)

    for mid, what, edits, witnesses, script, kind in MINE:
        row = run(mid, edits, witnesses, script, kind, based)
        row["property"] = what
        out["mutants"].append(row)

              flush=True)
        Path(OUT).write_text(json.dumps(out, indent=1) + "\n")
