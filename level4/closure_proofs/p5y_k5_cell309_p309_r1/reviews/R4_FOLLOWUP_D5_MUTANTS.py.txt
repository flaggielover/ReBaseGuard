def planted(edits):
    root = Path(tempfile.mkdtemp(prefix="m", dir=MUT))
    for d in ("code", "config", "tests", "verify"):
        shutil.copytree(FNS / d, root / d, ignore=shutil.ignore_patterns("__pycache__"))
    for rel, (how, text) in edits.items():
        p = root / rel
        if how == "new":
            p.write_text(text)
        else:
            p.write_text(p.read_text() + "\n" + text)
    scan = S.scan(root)
    static = SC.run(root)
    shutil.rmtree(root)
    kinds = sorted({f["kind"] for f in scan["findings"]})
    failed = sorted(k for k, v in static.items() if not v["pass"])
    return scan["verdict"], kinds, failed
