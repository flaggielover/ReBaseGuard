def my_hull(lo, hi, bits=10, n=4):
    s = 2 ** bits
    # floor/ceil by integer division only (independent of math.floor on Fraction)
    a = (lo.numerator * s) // lo.denominator
    b = -((-hi.numerator * s) // hi.denominator)
    L, H = Fr(a, s), Fr(b, s)
    if L == H:
        H = L + Fr(1, s)
    w = (H - L) / n
    return (L, H), [(L + j * w, L + (j + 1) * w) for j in range(n)]

def is_dyadic(x): d = Fr(x).denominator; return d & (d - 1) == 0
BAND = (Fr(6, 5), Fr(13, 5))
def banded(lo, hi):  # avoid ever passing band endpoints (not needed for arithmetic, but keep the firewall literal)
    return any(not (hi < a or lo > b) for a, b in (BAND, (-BAND[1], -BAND[0])))
